// Rename rules applied to libcompat.so so polyfill-glibc can drop its max
// glibc dep to <= 2.17 (well below the < 2.27 floor on
// pkg-build.racket-lang.org's test host).
//
// polyfill-glibc handles the bulk of the downgrade automatically (pthread_*
// 2.34 libc merge, vectorised libm expf/logf 2.27/2.29, stat/fstat, dlopen
// family, __isoc23_*, etc.).  The symbols below are the ones it doesn't know
// how to rewrite; we either redirect to an older glibc version or to a
// definition in libcompatshim.so shipped alongside.  Renames whose source
// symbol isn't present in libcompat.so are simply no-ops.

// __libc_single_threaded is a uint8_t flag added in 2.32, read by Rust std.
// Provide our own zero byte (value 0 forces the thread-safe path).
__libc_single_threaded@GLIBC_2.32 libcompatshim.so::__libc_single_threaded

// getentropy added in 2.25; Rust's getrandom prefers the syscall, so this
// reference (if present) is never exercised.  Stubbed in libcompatshim.so.
getentropy@GLIBC_2.25 libcompatshim.so::getentropy

// _Float128 conversions added in 2.26; stubbed in libcompatshim.so.
strfromf128@GLIBC_2.26 libcompatshim.so::strfromf128
strtof128@GLIBC_2.26  libcompatshim.so::strtof128

// lgamma/lgammaf/lgammal @2.23 only removed the write to the signgam global;
// the older default version is behaviorally equivalent for callers that don't
// read signgam.
lgamma@GLIBC_2.23  lgamma@GLIBC_2.2.5
lgammaf@GLIBC_2.23 lgammaf@GLIBC_2.2.5
lgammal@GLIBC_2.23 lgammal@GLIBC_2.2.5
